cmake_minimum_required(VERSION 3.22.1)
project(jassi_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)

add_subdirectory(whisper.cpp)

add_library(jassi_whisper SHARED whisper_jni.cpp)
target_include_directories(jassi_whisper PRIVATE whisper.cpp/include whisper.cpp/ggml/include)
target_link_libraries(jassi_whisper PRIVATE whisper android log)
target_link_options(jassi_whisper PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
